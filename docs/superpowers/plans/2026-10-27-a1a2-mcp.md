```python
PFX = "aabbccdd"
_BOOTSTRAPPED: set[Path] = set()


def _seed(db: Path, rid: str, title: str = "m-req") -> None:
    """需求行直插（指定 32-hex id）；created_at 存 naive UTC 字符串（UTCDateTime 口径）。
    new_db 是空文件无 schema：首个 _seed 先经 CLI `new`（子进程 ensure_schema）建表，
    bootstrap 行为随机 uuid4 需求，与本文件断言不冲突（前缀碰撞概率 2^-32）。"""
    if db not in _BOOTSTRAPPED:
        seed_req(db, "m-bootstrap")
        _BOOTSTRAPPED.add(db)
    conn = sqlite3.connect(str(db))
    conn.execute(
        "INSERT INTO requirements (id, source, title, acceptance, state, created_at, updated_at) "
        "VALUES (?, 'manual', ?, '[]', 'OPEN', '2026-01-01 00:00:00', '2026-01-01 00:00:00')",
        (rid, title),
    )
    conn.commit()
    conn.close()


def _snap(db: Path) -> tuple:
    """全局写入指纹（events 数, claims 数, 各需求 state）——零写入对照。"""
    ev = query(db, "SELECT COUNT(*) AS n FROM events")[0]["n"]
    cl = query(db, "SELECT COUNT(*) AS n FROM claims")[0]["n"]
    states = [(r["id"], r["state"]) for r in query(db, "SELECT id, state FROM requirements ORDER BY id")]
    return (ev, cl, tuple(states))


def _call_err2(m: MCP, name: str, args: dict) -> tuple[str, str]:
    """call + 解错误 payload 返回 (code, message)（ac_lib.call_err 不返 message，ac_cases_l 同模式）。"""
    res = m.call(name, args)
    if not res.get("isError"):
        raise AssertionError(f"{name} unexpectedly succeeded: {res}")
    payload = json.loads(res["content"][0]["text"])
    return payload["code"], payload.get("message", "")


def m_prefix_unique():
    print("== m1: 唯一 8 位前缀 -> 目标需求，事件落 32-hex（传播规则判别）")
    db = new_db()
    rid = "a1b2c3d4" + "f" * 24
    _seed(db, rid)
    m = MCP(db).start()
    try:
        m.call_ok("claim_task", {"requirement_id": rid[:8], "agent_name": "a"})
        ok("m1: 前缀 claim 成功", True, "")
        m.call_ok("add_record", {"requirement_id": rid[:8], "kind": "note", "title": "t"})
        ids = {r["requirement_id"] for r in query(db, "SELECT requirement_id FROM events")}
        ok("m1: 全部事件 requirement_id = 32-hex 全 id（传播到下游）", ids == {rid}, str(ids))
    finally:
        m.close()


def m_prefix_ambiguous():

    db = new_db()

    b = PFX + "0" * 23 + "2"
    _seed(db, a)
    _seed(db, b)
    m = MCP(db).start()
    try:

        code, msg = _call_err2(m, "claim_task", {"requirement_id": PFX, "agent_name": "x"})

        ok("m2: 消息含两候选", a in msg and b in msg, msg)

        ok("m2: 零写入（events/claims/state 不变）", _snap(db) == snap0, str(_snap(db)))
    finally:
        m.close()


def m_prefix_trunc():
```

```python
"""写保护（PRD 3.3）：agent 面写操作要求未结束认领（ended_at IS NULL），
校验与写入同事务（BEGIN IMMEDIATE 已保证写锁先行）。

requirement_id 解析单点（PRD 4.3；spec 2026-10-27-a1a2-mcp-surface §2.2）：
输入统一归一（strip / 大小写 / 连字符）；32 位 hex 精确匹配优先；
8~31 位 hex 唯一前缀可解析（>=2 命中 -> ambiguous_id）；<8 位或非 hex -> not_found。
MCP/REST/CLI 三面同源；前缀逻辑全仓仅存于本文件。"""

from __future__ import annotations

from sqlalchemy import String, func, select
from sqlalchemy.ext.asyncio import AsyncSession

from ..errors import BusinessError
from ..models import Claim, Requirement
from .ids import norm_id

_HEX = frozenset("0123456789abcdef")


def _not_found(raw: str) -> BusinessError:
    return BusinessError("not_found", f"requirement not found: {raw}", raw)


async def require_requirement(session: AsyncSession, requirement_id: str) -> Requirement:
    s = str(requirement_id).strip().lower().replace("-", "")
    is_hex = bool(s) and all(c in _HEX for c in s)
    if is_hex and len(s) == 32:
        req = await session.get(Requirement, s)
        if req is None:
            raise _not_found(requirement_id)
        return req
    if is_hex and 8 <= len(s) <= 31:
        # cast(String) 避 Uuid 列 .like 的 SADeprecationWarning（sqlalchemy 2.1.6 实测；
        # 未来版本将硬错 InvalidRequestError，盲审 P2-3）；SQLite 下 id 即 TEXT，CAST 无行为变化
        rows = (
            await session.scalars(
                select(Requirement.id)
                .where(Requirement.id.cast(String).like(f"{s}%"))
                .order_by(Requirement.id)
                .limit(6)
            )
        ).all()
        if not rows:
            raise _not_found(requirement_id)
        if len(rows) == 1:
            req = await session.get(Requirement, norm_id(rows[0]))
            if req is None:
                raise _not_found(requirement_id)
            return req
        total = (
            await session.scalar(
                select(func.count()).select_from(Requirement)
                .where(Requirement.id.cast(String).like(f"{s}%"))
            )
            or 0
        )
        cands = sorted(norm_id(r) for r in rows)
        shown = ", ".join(cands[:5]) + (" …" if len(cands) > 5 else "")
        raise BusinessError(
            "ambiguous_id",
            f"ambiguous requirement_id prefix '{s}': {shown}, 共 {total} 个候选; use a longer prefix",
            requirement_id,
        )
    raise _not_found(requirement_id)


async def get_active_claim(session: AsyncSession, requirement_id: str, requirement: Requirement | None = None) -> Claim:
    """同事务校验只查 ended_at IS NULL，不校验租约有效性（BLOCKED 期间仍可写）。"""
    if requirement is None:
        await require_requirement(session, requirement_id)
    claim = await session.scalar(
        select(Claim).where(Claim.requirement_id == requirement_id, Claim.ended_at.is_(None))
    )
    if claim is None:
        raise BusinessError("no_active_claim", f"no active claim for requirement {requirement_id}", requirement_id)
    return claim


async def count_active_claims(session: AsyncSession) -> int:
    return await session.scalar(
        select(func.count()).select_from(Claim).where(Claim.ended_at.is_(None))
    ) or 0
```
